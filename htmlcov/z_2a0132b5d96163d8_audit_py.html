1"""D27 — the audit trail: one writer, queryable with the agent stopped. 

2 

3Two tables in embedded SQLite: `runs` (one row per triage run) and 

4`events` (llm_call / tool_call / step / approval / cutoff, ordered by 

5seq). The store of record holds payloads — metrics and traces carry 

6hashes and sizes, the audit store carries the arguments and the ids — 

7so the incident question ("what did the agent look at before it 

8proposed this?") is answered from this file alone. If answering needs 

9the agent re-run, it's logs, not an audit trail. 

10 

11The harness feeds this through its `on_event` seam; the approval flow 

12and the step machine write their own kinds through the same writer. 

13SQLite over a composed service on purpose: the trail must outlive the 

14run and open anywhere, and laptop scale is the declared scale. 

15 

16The trail is tamper-EVIDENT, not tamper-proof: each event row hashes 

17the previous row's hash together with its own content (a per-run 

18chain), so an edited row, an inserted row, or a deletion before the 

19tail breaks every hash after it — `verify_chain` names the first 

20broken seq. The named residual: truncating the tail is silent, as in 

21any head-pointerless hash chain; the answer to that is a second 

22party holding the head, which D27 defers until there is one. 

23"""

83 self._conn.execute(

89 def finish_run(self, run_id: str, result: RunResult) -> None: 

90 verdicts = None

94 "suspects": len(result.report.suspects), 

95 "no_confident_candidate": result.report.no_confident_candidate,

104 result.usage.total_input, 

105 result.usage.output_tokens, 

106 int(result.degraded), 

107 verdicts, 

108 run_id,

113 def sink(self, run_id: str) -> Callable[[str, dict[str, Any]], None]: 

114 """The harness's `on_event` callback: stamps seq + timestamp, 

115 lifts latency_ms/tokens out of the payload into their columns."""

117 def write(kind: str, payload: dict[str, Any]) -> None: 

118 body = dict(payload) 

119 latency = body.pop("latency_ms", None) 

120 tokens = body.pop("tokens", None) 

121 self._append(run_id, kind, body, latency_ms=latency, tokens=tokens) 

122 

123 return write

140 def record_step_events( 

141 self, run_id: str, events: list[StepEvent], plan: list[PlannedStep] 

142 ) -> None: 

143 """StepEvents are progress surfaces and carry no arguments 

144 (D28); the audit row re-attaches the step's target so 

145 `--touched` answers from the store alone.""" 

146 for e in events: 

147 self._append( 

148 run_id, 

149 "step", 

150 { 

151 "tool": "apply_remediation", 

152 "action": e.action, 

153 "step_index": e.step_index, 

154 "status": e.status.value, 

155 "target": plan[e.step_index].target, 

156 "error": e.error, 

157 }, 

158 ts=e.timestamp, 

159 )

161 def _append( 

162 self, 

163 run_id: str, 

164 kind: str, 

165 payload: dict[str, Any], 

166 *, 

167 latency_ms: int | None = None, 

168 tokens: int | None = None, 

169 ts: datetime | None = None, 

170 ) -> None: 

171 row = self._conn.execute( 

172 "SELECT COALESCE(MAX(seq), -1) + 1 FROM events WHERE run_id=?", (run_id,) 

173 ).fetchone() 

174 prev = self._conn.execute( 

175 "SELECT row_hash FROM events WHERE run_id=? ORDER BY seq DESC LIMIT 1", (run_id,) 

176 ).fetchone() 

177 seq, body, when = row[0], json.dumps(payload), _iso(ts) or _now() 

178 digest = _chain_hash( 

179 prev[0] if prev else "", run_id, seq, kind, body, latency_ms, tokens, when 

180 ) 

181 self._conn.execute( 

182 "INSERT INTO events (run_id, seq, kind, payload, latency_ms, tokens, ts, row_hash)" 

183 " VALUES (?,?,?,?,?,?,?,?)", 

184 (run_id, seq, kind, body, latency_ms, tokens, when, digest), 

185 ) 

186 self._conn.commit()

190 def timeline(self, run_id: str) -> list[dict[str, Any]]: 

191 rows = self._conn.execute( 

192 "SELECT seq, kind, payload, latency_ms, tokens, ts FROM events" 

193 " WHERE run_id=? ORDER BY seq", 

194 (run_id,), 

195 ).fetchall() 

196 return [ 

197 { 

198 "seq": seq, 

199 "kind": kind, 

200 "payload": json.loads(payload), 

201 "latency_ms": latency_ms, 

202 "tokens": tokens, 

203 "ts": ts, 

204 } 

205 for seq, kind, payload, latency_ms, tokens, ts in rows 

206 ]

208 def touched(self, run_id: str) -> dict[str, list[str]]: 

209 """Distinct resources read (ids surfaced in tool results) and 

210 written (targets of steps that actually started).""" 

211 read: set[str] = set() 

212 written: set[str] = set() 

213 for event in self.timeline(run_id): 

214 p = event["payload"] 

215 if event["kind"] == "tool_call": 

216 read |= set(p.get("ids", ())) 

217 elif event["kind"] == "step" and p["status"] == StepStatus.STARTED.value: 

218 written.add(p["target"]) 

219 return {"read": sorted(read), "written": sorted(written)}

221 def tool_history(self, tool: str, *, since: timedelta | None = None) -> list[dict[str, Any]]: 

222 """Cross-run history for one tool; `apply_remediation` rows are 

223 the write history.""" 

224 if since is None: 

225 rows = self._conn.execute( 

226 "SELECT run_id, seq, kind, payload, ts FROM events" 

227 " WHERE json_extract(payload, '$.tool') = ? ORDER BY ts", 

228 (tool,), 

229 ).fetchall() 

230 else: 

231 rows = self._conn.execute( 

232 "SELECT run_id, seq, kind, payload, ts FROM events" 

233 " WHERE json_extract(payload, '$.tool') = ? AND ts >= ? ORDER BY ts", 

234 (tool, _iso(datetime.now(UTC) - since)), 

235 ).fetchall() 

236 return [ 

237 {"run_id": run_id, "seq": seq, "kind": kind, "payload": json.loads(payload), "ts": ts} 

238 for run_id, seq, kind, payload, ts in rows 

239 ]

241 def verify_chain(self, run_id: str) -> int | None: 

242 """Recompute the per-run hash chain; return the first seq whose 

243 stored hash disagrees, or None if the trail proves itself.""" 

244 rows = self._conn.execute( 

245 "SELECT seq, kind, payload, latency_ms, tokens, ts, row_hash FROM events" 

246 " WHERE run_id=? ORDER BY seq", 

247 (run_id,), 

248 ).fetchall() 

249 prev = "" 

250 for seq, kind, payload, latency_ms, tokens, ts, row_hash in rows: 

251 if _chain_hash(prev, run_id, seq, kind, payload, latency_ms, tokens, ts) != row_hash: 

252 return int(seq) 

253 prev = row_hash 

254 return None

256 def run_row(self, run_id: str) -> dict[str, Any] | None: 

257 row = self._conn.execute("SELECT * FROM runs WHERE run_id=?", (run_id,)).fetchone() 

258 if row is None: 

259 return None 

260 cols = [c[0] for c in self._conn.execute("SELECT * FROM runs LIMIT 0").description] 

261 return dict(zip(cols, row, strict=True))

264def parse_since(text: str) -> timedelta: 

265 """`7d`, `24h`, `30m` — the CLI's --since grammar.""" 

266 units = {"d": "days", "h": "hours", "m": "minutes"} 

267 unit = text[-1:] 

268 if unit not in units or not text[:-1].isdigit(): 

269 raise ValueError(f"--since wants <n>d|<n>h|<n>m, got {text!r}") 

270 return timedelta(**{units[unit]: int(text[:-1])})

273def _chain_hash( 

274 prev: str, 

275 run_id: str, 

276 seq: int, 

277 kind: str, 

278 payload: str, 

279 latency_ms: int | None, 

280 tokens: int | None, 

281 ts: str, 

282) -> str: 

283 material = json.dumps([prev, run_id, seq, kind, payload, latency_ms, tokens, ts]) 

284 return hashlib.sha256(material.encode()).hexdigest()
