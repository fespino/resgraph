1"""The registry's third surface: Anthropic tool blocks, executed in-process. 

2 

3Same derivation discipline as the MCP and HTTP surfaces — TOOL_REGISTRY 

4is the only source, and this surface additionally refuses anything that 

5is not read-only: the agent's surface is safe by construction, not by 

6prompt. 

7 

8The refusal is a SESSION composition rule, not a per-tool one (D26): 

9this session reads platform data, so no open-world tool (untrusted 

10content in) and no write-capable tool (a channel out) may join it — 

11holding all three at once is the lethal trifecta. Risk lives in the 

12combination; each tool alone can be justified. 

13"""

43def agent_entries( 

44 entries: tuple[ToolRegistration, ...] = TOOL_REGISTRY, 

45) -> tuple[ToolRegistration, ...]: 

46 """The registry subset an agent session may hold. Selection, not 

47 exclusion: a tool joins the surface by satisfying every condition, 

48 so a new registration is off the agent's surface until it earns a 

49 place there.""" 

50 return tuple( 

51 e for e in entries if e.hints.read_only and not e.privileged and not e.hints.open_world 

52 )

55class RegistryToolset: 

56 def __init__( 

57 self, 

58 qctx_factory: Callable[[], QueryContext], 

59 entries: tuple[ToolRegistration, ...] | None = None, 

60 ) -> None: 

61 entries = agent_entries() if entries is None else entries 

62 for entry in entries: 

63 if not entry.hints.read_only or entry.privileged: 

64 raise RuntimeError( 

65 f"tool {entry.name!r} is not a plain read tool; the analyst surface refuses it" 

66 ) 

67 if entry.hints.open_world: 

68 raise RuntimeError( 

69 f"tool {entry.name!r} is open-world; a session reading platform " 

70 "data admits no untrusted-content channel (D26 composition rule)" 

71 ) 

72 self._entries = {e.name: e for e in entries} 

73 self._qctx_factory = qctx_factory 

74 

75 def blocks(self) -> list[dict[str, Any]]: 

76 return [ 

77 { 

78 "name": e.name, 

79 "description": e.description, 

80 "input_schema": e.input_model.model_json_schema(), 

81 } 

82 for e in self._entries.values() 

83 ] 

84 

85 def execute(self, name: str, args: dict[str, Any]) -> ToolOutcome: 

86 entry = self._entries.get(name) 

87 if entry is None: 

88 return _error("invalid_input", f"unknown tool {name!r}", "rephrase") 

89 try: 

90 parsed = entry.input_model.model_validate(args) 

91 except ValidationError as e: 

92 details = "; ".join( 

93 f"{'.'.join(str(p) for p in err['loc'])}: {err['msg']}" for err in e.errors() 

94 ) 

95 return _error("invalid_input", details, entry.error_actions["invalid_input"]) 

96 qctx = self._qctx_factory() 

97 try: 

98 result = entry.fn(parsed, ctx=CallerContext("analyst", entry.scopes, qctx)) 

99 return ToolOutcome(ok=True, payload=result.model_dump_json()) 

100 except Exception as e: 

101 return _error("store_unavailable", str(e), entry.error_actions["store_unavailable"]) 

102 finally: 

103 if qctx.session is not None: 

104 qctx.session.close()
