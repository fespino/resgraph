1"""Compare labeled arms of the same suite: cost, latency, and pass rate 

2per arm. 

3 

4`report` aggregates one run and `gate` compares a run to the certified 

5baseline; neither answers "which worker earns its price." An arm is a 

6run of the same item set under one changed variable (worker model, or 

7the skill on/off). The money metric is worker cost divided by the 

8triages that actually passed at k; latency rides alongside it, because a 

9cheaper worker that is far slower is a different trade, not a free win. 

10""" 

11 

12from collections import defaultdict 

13from typing import Any

15from .report import aggregate, item_passed 

16from .runner import estimate_cost

19def arm_summary(label: str, rows: list[dict[str, Any]]) -> dict[str, Any]: 

20 agg = aggregate(rows) 

21 worker_cost = sum(estimate_cost(r["tokens"], r["model"]) for r in rows) 

22 passed_items = sum(all(v) for v in _by_item(rows).values()) 

23 return { 

24 "label": label, 

25 "items": agg["items"], 

26 "item_ids": agg["item_ids"], 

27 "trials": agg["trials"], 

28 "pass_all_trials": agg["pass_all_trials"], 

29 "pass_any_trial": agg["pass_any_trial"], 

30 "fabrication_count": agg["fabrication_count"], 

31 "slices": agg["slices"], 

32 "worker_cost": worker_cost, 

33 "passed_items": passed_items, 

34 "cost_per_passed": worker_cost / passed_items if passed_items else None, 

35 "latency_p50_s": agg["latency_p50_s"], 

36 "latency_p95_s": agg["latency_p95_s"], 

37 "models": sorted({r["model"] for r in rows if r.get("model")}), 

38 }

41def _by_item(rows: list[dict[str, Any]]) -> dict[str, list[bool]]: 

42 by_item: dict[str, list[bool]] = defaultdict(list) 

43 for row in rows: 

44 by_item[row["scenario_id"]].append(item_passed(row)) 

45 return by_item

48def compare(arms: list[dict[str, Any]], baseline: str) -> dict[str, Any]: 

49 """Deltas against the baseline arm. Declines if the arms do not 

50 cover the same item set — a cost comparison across different work 

51 is meaningless, the same rule the gate applies to baselines.""" 

52 base = next((a for a in arms if a["label"] == baseline), None) 

53 if base is None: 

54 raise ValueError(f"baseline arm {baseline!r} not among {[a['label'] for a in arms]}") 

55 base_ids = set(base["item_ids"]) 

56 mismatched = [a["label"] for a in arms if set(a["item_ids"]) != base_ids] 

57 comparable = not mismatched 

58 deltas = {} 

59 for a in arms: 

60 if a["label"] == baseline: 

61 continue 

62 d = None 

63 if a["pass_all_trials"] is not None and base["pass_all_trials"] is not None: 

64 d = a["pass_all_trials"] - base["pass_all_trials"] 

65 deltas[a["label"]] = d 

66 return { 

67 "baseline": baseline, 

68 "comparable": comparable, 

69 "mismatched": mismatched, 

70 "deltas": deltas, 

71 }

74def _fmt(v: float | None, places: int = 3) -> str: 

75 return "—" if v is None else f"{v:.{places}f}"

78def render(arms: list[dict[str, Any]], comparison: dict[str, Any]) -> str: 

79 lines = [ 

80 f"arms={len(arms)} baseline={comparison['baseline']} comparable={comparison['comparable']}", 

81 ] 

82 if not comparison["comparable"]: 

83 lines.append(f" DECLINED: item sets differ ({', '.join(comparison['mismatched'])})") 

84 header = ( 

85 f" {'arm':<10} {'pass^k':>7} {'Δ':>7} {'cost$':>8} {'$/passed':>9} " 

86 f"{'p50s':>6} {'p95s':>6} {'fab':>4}" 

87 ) 

88 lines.append(header) 

89 for a in arms: 

90 d = comparison["deltas"].get(a["label"]) 

91 lines.append( 

92 f" {a['label']:<10} {_fmt(a['pass_all_trials'], 2):>7} " 

93 f"{('' if d is None else f'{d:+.2f}'):>7} {_fmt(a['worker_cost'], 2):>8} " 

94 f"{_fmt(a['cost_per_passed'], 3):>9} {_fmt(a['latency_p50_s'], 1):>6} " 

95 f"{_fmt(a['latency_p95_s'], 1):>6} {a['fabrication_count']:>4}" 

96 ) 

97 return "\n".join(lines)
