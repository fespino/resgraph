1"""Filter DSL: conjunctions of comparisons, nothing else (D16). 

2 

3`type=vm AND attrs.zone=z1 AND attrs.cpu>=4` parses into Predicates; 

4OR, grouping, and functions are rejected at the boundary. Parse, don't 

5validate: downstream code only ever sees well-formed Predicates. 

6""" 

7 

8import re 

9from dataclasses import dataclass 

10from typing import Literal, cast 

11 

12Op = Literal["=", "!=", "<", "<=", ">", ">="] 

13 

14_TERM = re.compile( 

15 r"^(?P<field>[A-Za-z_][A-Za-z0-9_.]*)\s*(?P<op><=|>=|!=|=|<|>)\s*(?P<value>\S+)$" 

16) 

17_AND = re.compile(r"\bAND\b", re.IGNORECASE) 

18 

19MAX_FILTER_LEN = 512 

20 

21 

22@dataclass(frozen=True) 

23class Predicate: 

24 field: str 

25 op: Op 

26 value: str | int | float | bool 

27 

28 def __str__(self) -> str: 

29 return f"{self.field}{self.op}{self.value}" 

30 

31 

32def _coerce(raw: str) -> str | int | float | bool: 

33 if raw.lower() in ("true", "false"): 

34 return raw.lower() == "true" 

35 try: 

36 return int(raw) 

37 except ValueError: 

38 pass 

39 try: 

40 return float(raw) 

41 except ValueError: 

42 return raw 

43 

44 

45def parse_filter(text: str | None) -> list[Predicate]: 

46 """Parse a filter string into predicates; raise ValueError on anything 

47 outside the D16 grammar.""" 

48 if not text or not text.strip(): 

49 return [] 

50 if len(text) > MAX_FILTER_LEN: 

51 raise ValueError( 

52 f"filter longer than {MAX_FILTER_LEN} chars — drop redundant terms " 

53 "or split into separate calls" 

54 ) 

55 if re.search(r"\bOR\b", text, re.IGNORECASE): 

56 raise ValueError( 

57 "OR is not supported — make one call per branch and merge the " 

58 "results, or use a single AND chain: type=vm AND attrs.zone=z1" 

59 ) 

60 if "(" in text or ")" in text: 

61 raise ValueError( 

62 "parentheses are not supported — write a flat AND chain: type=vm AND attrs.zone=z1" 

63 ) 

64 preds = [] 

65 for term in (t.strip() for t in _AND.split(text.strip())): 

66 m = _TERM.match(term) 

67 if not m: 

68 raise ValueError( 

69 f"cannot parse filter term: {term!r} — terms look like " 

70 "type=vm or attrs.cpu>=4, joined with AND" 

71 ) 

72 value = _coerce(m["value"]) 

73 op = cast(Op, m["op"]) # _TERM's op group closes exactly this set 

74 if op in ("<", "<=", ">", ">=") and isinstance(value, str): 

75 raise ValueError( 

76 f"ordering comparison needs a numeric value: {term!r} — use = " 

77 "or != for strings, or compare a numeric attr like attrs.cpu>=4" 

78 ) 

79 preds.append(Predicate(m["field"], op, value)) 

80 return preds
